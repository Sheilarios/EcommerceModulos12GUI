cmake_minimum_required(VERSION 3.15)
project(EcommerceAmazon CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
include_directories(include)
# Version de consola (Windows, Linux, macOS)
add_executable(ecommerce_consola src/consola.cpp)
# Interfaz grafica (solo Windows)
if(WIN32)
  add_executable(EcommerceModulos12GUI WIN32 src/main.cpp)
  target_link_libraries(EcommerceModulos12GUI comctl32 comdlg32)
  if(MSVC)
    target_compile_options(EcommerceModulos12GUI PRIVATE /utf-8)
  else()
    target_link_options(EcommerceModulos12GUI PRIVATE -municode)
  endif()
endif()
